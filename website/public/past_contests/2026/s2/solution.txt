#include <bits/stdc++.h>
using namespace std;

int main() {
  int n, l, q;
  cin >> n >> l >> q;

  vector<int> c(n + 1, 0);
  vector<bool> light(n + 1, false);
  for (int i = 0; i < l; i++) {
    int p, s;
    cin >> p >> s;
    int start = p - s, end = p + s;
    if (start < 1) {
      start = 1;
    }
    c[start]++;
    if (end < n) {
      c[end + 1]--;
    }
  }

  int r = 0;
  for (int i = 1; i < n + 1; i++) {
    r += c[i];
    if (r > 0) {
      light[i] = true;
    }
  }

  for (int i = 0; i < q; i++) {
    int p;
    cin >> p;
    if (light[p]) {
      cout << "Y" << endl;
    } else {
      cout << "N" << endl;
    }
  }
}
